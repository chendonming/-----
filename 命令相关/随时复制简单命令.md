# 随时复制简单命令

```bash
# java springboot启动
mvn spring-boot:run

# 编译
mvn compile

# 运行（指定主类）
mvn exec:java -Dexec.mainClass="com.example.Main"

# 组合拳: 清理 + 编译 + 测试 + 打包 + 安装
mvn clean install

# 扫描文件夹下的文件，将其转为utf-8
find /Users/chendongmin/Documents/workplace/mall/mall-single-app -name "*.java" -type f -exec perl -pi -e 's/^\xEF\xBB\xBF//' {} \+

# 获取上周一到周日的git提交日志
git log --since="monday last week" --until="monday this week" --pretty=format:"%h - %an, %ad : %s" --author="chendm" > output.txt
```
